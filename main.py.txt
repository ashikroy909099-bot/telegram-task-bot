from keep_alive import keep_alive
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import (ApplicationBuilder, CommandHandler,
                          CallbackQueryHandler, MessageHandler, ContextTypes,
                          filters)
from database import (add_user, get_user, get_tasks, get_available_tasks,
                      get_task, add_task, delete_task, update_balance,
                      add_withdraw, get_withdraws, add_report, approve_report,
                      check_report_exists)

# ---------- CONFIGURATION ----------
TOKEN = "8265605610:AAFbtXYbdiZTxoUXMkPIfsl_M-osBZ2-P2s"
ADMIN_ID = 6121567598
ALLOWED_USERS = [5158578932, 7022463045, 8018554461]
MIN_WITHDRAW = 1.0  # USD

# ---------- STATES ----------
waiting_for_report = {}  # user_id: task_id


# ---------- START ----------
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    uid = update.effective_user.id
    if uid not in ALLOWED_USERS and uid != ADMIN_ID:
        await update.message.reply_text("🚫 Access denied.")
        return

    add_user(uid)
    kb = [[InlineKeyboardButton("📌 Tasks", callback_data="tasks")],
          [InlineKeyboardButton("📊 My Report", callback_data="report_stats")],
          [InlineKeyboardButton("💰 Balance", callback_data="balance")],
          [InlineKeyboardButton("💸 Withdraw", callback_data="withdraw")]]
    if uid == ADMIN_ID:
        kb.append(
            [InlineKeyboardButton("👑 Admin Panel", callback_data="admin")])

    await update.message.reply_text("✅ Welcome to the Task Bot",
                                    reply_markup=InlineKeyboardMarkup(kb))


# ---------- BUTTONS ----------
async def buttons(update: Update, context: ContextTypes.DEFAULT_TYPE):
    q = update.callback_query
    await q.answer()
    uid = q.from_user.id
    user = get_user(uid)

    if q.data == "tasks":
        tasks = get_available_tasks(uid)
        if not tasks:
            await q.message.reply_text("No new tasks available.")
            return

        for t in tasks:
            kb = [[
                InlineKeyboardButton("📤 Submit Report",
                                     callback_data=f"submit_report|{t[0]}")
            ]]
            msg = f"#{t[0]} {t[1]}\n📝 {t[2]}\n💵 Reward: $ {t[3]:.2f}"
            await q.message.reply_text(msg,
                                       reply_markup=InlineKeyboardMarkup(kb))

    elif q.data.startswith("submit_report|"):
        task_id = int(q.data.split("|")[1])
        waiting_for_report[uid] = task_id
        await q.message.reply_text(
            f"📤 Report for Task #{task_id}\nPlease send screenshot or text proof now."
        )

    elif q.data == "report_stats":
        await q.message.reply_text(
            f"⏳ Pending: {user[2]}\n✅ Completed: {user[3]}")

    elif q.data == "balance":
        await q.message.reply_text(f"💰 Balance: $ {user[1]:.2f}")

    elif q.data == "withdraw":
        if user[1] < MIN_WITHDRAW:
            await q.message.reply_text(
                f"❌ Minimum withdraw is $ {MIN_WITHDRAW:.2f}")
            return
        context.user_data["withdraw_step"] = 1
        await q.message.reply_text("💵 Enter withdraw amount (USD)")

    elif q.data == "admin" and uid == ADMIN_ID:
        await q.message.reply_text("👑 Admin Commands:\n\n"
                                   "/addtask | Title | Description | Reward\n"
                                   "/notice TEXT\n"
                                   "/deposit user_id amount\n"
                                   "/approve user_id task_id\n"
                                   "/viewwithdraw\n"
                                   "/balance user_id\n"
                                   "/tasks\n"
                                   "/deltask task_id")


# ---------- HANDLERS ----------
async def receive_report(update: Update, context: ContextTypes.DEFAULT_TYPE):
    uid = update.effective_user.id
    if uid not in waiting_for_report: return

    task_id = waiting_for_report[uid]

    if check_report_exists(uid, task_id):
        await update.message.reply_text("⚠️ You already submitted this task.")
        del waiting_for_report[uid]
        return

    proof_text = update.message.text if update.message.text else "Screenshot attached"

    if add_report(uid, task_id, proof_text):
        caption = f"📥 New Task Submission\n\n👤 User ID: {uid}\n🆔 Task ID: #{task_id}"
        if update.message.photo:
            await context.bot.send_photo(
                ADMIN_ID,
                photo=update.message.photo[-1].file_id,
                caption=caption)
        else:
            await context.bot.send_message(
                ADMIN_ID, f"{caption}\n\n📝 Proof:\n{proof_text}")

        await update.message.reply_text("✅ Task submitted successfully!")
    else:
        await update.message.reply_text("❌ Submission failed (Internal Error)."
                                        )

    del waiting_for_report[uid]


# ---------- ADMIN COMMANDS ----------
async def balance_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != ADMIN_ID: return
    try:
        user_id = int(context.args[0])
        user = get_user(user_id)
        if not user:
            await update.message.reply_text("❌ User not found")
            return
        await update.message.reply_text(
            f"👤 User: {user_id}\n💰 Balance: $ {user[1]:.2f}\n⏳ Pending: {user[2]}\n✅ Completed: {user[3]}"
        )
    except:
        await update.message.reply_text("❌ Use: /balance user_id")


async def tasks_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != ADMIN_ID: return
    tasks = get_tasks()
    if not tasks:
        await update.message.reply_text("❌ No tasks found.")
        return
    msg = "📋 Task List:\n\n"
    for t in tasks:
        msg += f"#{t[0]} {t[1]} – $ {t[3]:.2f}\n"
    await update.message.reply_text(msg)


async def deltask_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != ADMIN_ID: return
    try:
        task_id = int(context.args[0])
        task = get_task(task_id)
        if not task:
            await update.message.reply_text("❌ Task not found")
            return
        delete_task(task_id)
        await update.message.reply_text(f"🗑 Task #{task_id} deleted")
    except:
        await update.message.reply_text("❌ Use: /deltask task_id")


# ---------- MESSAGE ----------
async def admin_message_handler(update: Update,
                                context: ContextTypes.DEFAULT_TYPE):
    uid = update.effective_user.id
    text = update.message.text
    if uid != ADMIN_ID or not text: return

    if text.startswith("/addtask"):
        try:
            _, data = text.split(" ", 1)
            parts = [p.strip() for p in data.split("|")]
            if len(parts) == 4:
                add_task(parts[1], parts[2], float(parts[3]))
                await update.message.reply_text("✅ Task added successfully")
        except:
            await update.message.reply_text(
                "❌ Format: /addtask | Title | Description | Reward")

    elif text.startswith("/notice"):
        msg = text.replace("/notice ", "")
        for u in ALLOWED_USERS:
            try:
                await context.bot.send_message(u, f"📢 Announcement:\n{msg}")
            except:
                pass
        await update.message.reply_text("✅ Notice sent.")

    elif text.startswith("/deposit"):
        try:
            _, u, amt = text.split()
            update_balance(int(u), float(amt))
            await context.bot.send_message(int(u), f"💰 Deposit: $ {amt}")
            await update.message.reply_text("✅ Deposit successful.")
        except:
            await update.message.reply_text("❌ Format: /deposit user_id amount"
                                            )

    elif text.startswith("/approve"):
        try:
            _, u, t = text.split()
            u, t = int(u), int(t)
            task = get_task(t)
            reward = task[3]
            approve_report(u, t, reward)
            await context.bot.send_message(
                u, f"✅ Task Approved\n💵 $ {reward:.2f} added to your balance")
            await update.message.reply_text("✔ Approved.")
        except:
            await update.message.reply_text(
                "❌ Format: /approve user_id task_id")

    elif text == "/viewwithdraw":
        data = get_withdraws()
        if not data: await update.message.reply_text("No withdraws.")
        else:
            for w in data:
                await update.message.reply_text(
                    f"User: {w[1]}\nAmount: $ {w[2]:.2f}\nAddress: {w[3]}")


# ---------- WITHDRAW FLOW ----------
async def withdraw_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    uid = update.effective_user.id
    text = update.message.text
    if "withdraw_step" not in context.user_data: return

    if context.user_data["withdraw_step"] == 1:
        try:
            context.user_data["amount"] = float(text)
            context.user_data["withdraw_step"] = 2
            await update.message.reply_text("🏦 Enter payment address")
        except:
            await update.message.reply_text("❌ Invalid amount.")
    elif context.user_data["withdraw_step"] == 2:
        amount = context.user_data["amount"]
        add_withdraw(uid, amount, text)
        context.user_data.clear()
        await update.message.reply_text("📩 Withdraw request submitted.")
        await context.bot.send_message(
            ADMIN_ID,
            f"💸 Withdraw Request\nUser: {uid}\nAmount: $ {amount:.2f}\nAddress: {text}"
        )


# ---------- RUN ----------
if __name__ == "__main__":
    keep_alive()
    app = ApplicationBuilder().token(TOKEN).build()
    app.add_handler(CommandHandler("balance", balance_cmd))
    app.add_handler(CommandHandler("tasks", tasks_cmd))
    app.add_handler(CommandHandler("deltask", deltask_cmd))
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CallbackQueryHandler(buttons))
    app.add_handler(MessageHandler(filters.PHOTO | filters.TEXT,
                                   receive_report),
                    group=1)
    app.add_handler(MessageHandler(filters.TEXT & filters.Regex("^/"),
                                   admin_message_handler),
                    group=2)
    app.add_handler(MessageHandler(filters.TEXT & ~filters.Regex("^/"),
                                   withdraw_handler),
                    group=3)
    print("Bot is running...")
    app.run_polling()
